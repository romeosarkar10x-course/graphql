"""
Writes, Module 4. Every mutation takes one `input` argument and returns a payload (4.2), except
`uploadCover`, whose `file` must be a separate argument for the multipart request convention (4.6).
"""
type Mutation {
  "Signed-URL alternative, step 2: tell Shelfie the upload finished; it checks the object and links it."
  confirmCoverUpload(input: ConfirmCoverUploadInput!): UploadCoverPayload!
  "Signed-URL alternative (4.6), step 1: get a short-lived URL and PUT the bytes to object storage."
  createCoverUpload(input: CreateCoverUploadInput!): CreateCoverUploadPayload!
  "Payload style: rule violations come back in `userErrors`, never as top-level errors (4.3)."
  createReview(input: CreateReviewInput!): CreateReviewPayload!
  "Result-union style: the same rules as `createReview`, one member type per outcome (4.3)."
  createReviewTyped(input: CreateReviewInput!): CreateReviewResult!
  "Only the review's author may delete it."
  deleteReview(input: DeleteReviewInput!): DeleteReviewPayload!
  "Joining a club you are already in succeeds and changes nothing: the mutation is idempotent (4.4)."
  joinClub(input: JoinClubInput!): JoinClubPayload!
  "Leaving a club you are not in succeeds and changes nothing. The owner cannot leave."
  leaveClub(input: LeaveClubInput!): LeaveClubPayload!
  """
  GraphQL multipart request convention (4.6): send `multipart/form-data` with `operations`, `map` and
  the file parts, plus a `GraphQL-Require-Preflight: 1` header (CSRF protection). The server holds the
  whole file in memory, which `serverBufferedBytes` reports.
  """
  uploadCover(bookId: ID!, file: Upload!): UploadCoverPayload!
}

"""
Shelfie reference schema, Modules 0 to 4. Deliberately small: timestamps are plain Strings until

## Operations the API's owners describe for agents (you may copy and adapt them)
```graphql
"""
One book by its id ($id, for example b1; get ids from FindBooks or Search): title, authors, year, page
count, synopsis, mean rating (null without reviews) and its three newest reviews with each reviewer's handle.
Returns null for an unknown id.
"""
query BookDetails($id: ID!) {
  book(id: $id) {
    id
    title
    authors
    publishedYear
    pageCount
    synopsis
    averageRating
    reviews(last: 3) {
      rating
      body
      author {
        handle
      }
    }
  }
}
```
```graphql
"""
One club by its id ($id, for example c1; get ids from ListClubs or Search): description, visibility, its
owner (name and handle), member count, the first 10 members in join order and the current book. Returns null for
an unknown id.
"""
query ClubDetails($id: ID!) {
  club(id: $id) {
    id
    name
    description
    visibility
    owner {
      id
      name
      handle
    }
    memberCount
    members(first: 10) {
      id
      name
      handle
    }
    currentBook {
      id
      title
    }
  }
}
```
```graphql
"""
The meetups of one club ($clubId, for example c1): title, start time (UTC, ISO 8601) and location. By
default only upcoming meetups; set $upcomingOnly to false to include past ones, for example to answer "when was
the last meetup".
"""
query ClubMeetups($clubId: ID!, $upcomingOnly: Boolean = true) {
  club(id: $clubId) {
    id
    name
    meetups(upcomingOnly: $upcomingOnly) {
      title
      startsAt
      location
    }
  }
}
```
```graphql
"""
Publish a review as the signed-in reader: $bookId (a book id such as b2), $rating from 1 to 5 and $body
of 10 to 2000 characters. Changes data and is public at once; a reader can review a book only once (DUPLICATE).
Rule violations come back in userErrors with the input field, a code and a message.
"""
mutation CreateReview($bookId: ID!, $rating: Int!, $body: String!) {
  createReview(input: {bookId: $bookId, rating: $rating, body: $body}) {
    review {
      id
      rating
      body
    }
    userErrors {
      field
      code
      message
    }
  }
}
```
```graphql
"""
Find books in the Shelfie catalogue whose title contains $titleContains (ignoring case); leave it out to list
the catalogue in order. Returns at most $first books (default 5, the server caps nothing, keep it small) with
authors, publication year and mean rating. Use it to turn a title into a book id. There is no author filter: to
find an author, list the catalogue and filter yourself.
"""
query FindBooks($titleContains: String, $first: Int = 5) {
  books(titleContains: $titleContains, first: $first) {
    id
    title
    authors
    publishedYear
    averageRating
  }
}
```
```graphql
"""
Make the signed-in reader a member of the club $clubId. Changes data: the reader becomes visible to the
club. Joining a club twice succeeds and changes nothing. Problems (not signed in, unknown club) come back in
userErrors with a code, not as an error.
"""
mutation JoinClub($clubId: ID!) {
  joinClub(input: {clubId: $clubId}) {
    club {
      id
      name
      memberCount
    }
    userErrors {
      code
      message
    }
  }
}
```
```graphql
"""
Every book club on Shelfie with its visibility (PUBLIC or PRIVATE), member count and the book it is
reading right now, with that book's mean rating. No arguments; there are only a few clubs. Start here for any
question that compares clubs.
"""
query ListClubs {
  clubs {
    id
    name
    visibility
    memberCount
    currentBook {
      id
      title
      averageRating
    }
  }
}
```
```graphql
"""
One reader by user id ($id, for example u1; get ids from Search, which matches names and handles):
name, handle, bio, location and the clubs they belong to. Returns null for an unknown id.
"""
query ReaderProfile($id: ID!) {
  user(id: $id) {
    id
    name
    handle
    bio
    location
    clubs {
      id
      name
    }
  }
}
```
```graphql
"""
Every review one reader ($id, a user id such as u1) has written: rating from 1 to 5, text, when it was
written and which book it is about. Use ReaderProfile or Search first if you only know a name.
"""
query ReaderReviews($id: ID!) {
  user(id: $id) {
    id
    handle
    reviews {
      rating
      body
      createdAt
      book {
        id
        title
      }
    }
  }
}
```
```graphql
"""
Search Shelfie for $text, ignoring case, across book titles, club names and reader names and handles.
Returns books first, then clubs, then readers, each with its __typename and id, so a follow-up tool can fetch
details. The quickest way to turn a name into an id.
"""
query Search($text: String!) {
  search(text: $text) {
    __typename
    ... on Book {
      id
      title
    }
    ... on Club {
      id
      name
    }
    ... on User {
      id
      name
      handle
    }
  }
}
```
